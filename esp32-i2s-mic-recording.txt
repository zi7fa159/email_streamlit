#include <Arduino.h>
#include <driver/i2s.h>
#include <SPIFFS.h>

#define I2S_PORT      I2S_NUM_0
#define I2S_WS        15
#define I2S_SCK       14
#define I2S_SD        32
#define SAMPLE_RATE   16000
#define SAMPLE_BITS   16
#define BYTES_PER_SAMPLE (SAMPLE_BITS / 8)
#define WAV_HDR_SIZE  44
#define RECORD_TIME   5
#define BUF_SIZE      512
#define FILENAME      "/rec.wav"

void createWavHeader(uint8_t *hdr, uint32_t dataSize) {
  uint32_t fileSize = dataSize + WAV_HDR_SIZE - 8;
  hdr[0] = 'R'; hdr[1] = 'I'; hdr[2] = 'F'; hdr[3] = 'F';
  hdr[4] = fileSize & 0xFF; hdr[5] = (fileSize >> 8) & 0xFF; hdr[6] = (fileSize >> 16) & 0xFF; hdr[7] = (fileSize >> 24) & 0xFF;
  hdr[8] = 'W'; hdr[9] = 'A'; hdr[10] = 'V'; hdr[11] = 'E';
  hdr[12] = 'f'; hdr[13] = 'm'; hdr[14] = 't'; hdr[15] = ' ';
  hdr[16] = 16; hdr[17] = 0; hdr[18] = 0; hdr[19] = 0;
  hdr[20] = 1; hdr[21] = 0; hdr[22] = 1; hdr[23] = 0;
  hdr[24] = SAMPLE_RATE & 0xFF; hdr[25] = (SAMPLE_RATE >> 8) & 0xFF; hdr[26] = (SAMPLE_RATE >> 16) & 0xFF; hdr[27] = (SAMPLE_RATE >> 24) & 0xFF;
  uint32_t br = SAMPLE_RATE * BYTES_PER_SAMPLE;
  hdr[28] = br & 0xFF; hdr[29] = (br >> 8) & 0xFF; hdr[30] = (br >> 16) & 0xFF; hdr[31] = (br >> 24) & 0xFF;
  hdr[32] = BYTES_PER_SAMPLE; hdr[33] = 0;
  hdr[34] = SAMPLE_BITS; hdr[35] = 0;
  hdr[36] = 'd'; hdr[37] = 'a'; hdr[38] = 't'; hdr[39] = 'a';
  hdr[40] = dataSize & 0xFF; hdr[41] = (dataSize >> 8) & 0xFF; hdr[42] = (dataSize >> 16) & 0xFF; hdr[43] = (dataSize >> 24) & 0xFF;
}

void i2s_init() {
  i2s_driver_uninstall(I2S_PORT);
  i2s_config_t cfg = {
    .mode = (i2s_mode_t)(I2S_MODE_MASTER | I2S_MODE_RX),
    .sample_rate = SAMPLE_RATE,
    .bits_per_sample = (i2s_bits_per_sample_t)SAMPLE_BITS,
    .channel_format = I2S_CHANNEL_FMT_ONLY_LEFT,
    .communication_format = I2S_COMM_FORMAT_I2S,
    .intr_alloc_flags = ESP_INTR_FLAG_LEVEL1,
    .dma_buf_count = 4,
    .dma_buf_len = BUF_SIZE,
    .use_apll = false
  };
  i2s_pin_config_t pins = {
    .bck_io_num = I2S_SCK,
    .ws_io_num = I2S_WS,
    .data_out_num = I2S_PIN_NO_CHANGE,
    .data_in_num = I2S_SD
  };
  if(i2s_driver_install(I2S_PORT, &cfg, 0, NULL) != ESP_OK) { Serial.println("I2S install error"); return; }
  if(i2s_set_pin(I2S_PORT, &pins) != ESP_OK) { Serial.println("I2S pin error"); return; }
  i2s_start(I2S_PORT);
}

void listFiles() {
  File root = SPIFFS.open("/");
  File f = root.openNextFile();
  Serial.println("SPIFFS files:");
  while(f) {
    Serial.printf("  %s (%u bytes)\n", f.name(), f.size());
    f = root.openNextFile();
  }
}

bool checkFreeSpace(size_t req) {
  size_t free = SPIFFS.totalBytes() - SPIFFS.usedBytes();
  if(free < req) { Serial.printf("Not enough space. Req: %u, Avail: %u\n", req, free); return false; }
  return true;
}

bool recordAudio() {
  size_t estSize = RECORD_TIME * SAMPLE_RATE * BYTES_PER_SAMPLE;
  size_t totalReq = estSize + WAV_HDR_SIZE;
  if(!checkFreeSpace(totalReq)) {
    Serial.println("Formatting SPIFFS...");
    SPIFFS.format();
    if(!SPIFFS.begin(true) || !checkFreeSpace(totalReq)) {
      Serial.println("Space still insufficient.");
      return false;
    }
  }
  File f = SPIFFS.open(FILENAME, "w");
  if(!f) { Serial.println("File open error"); return false; }
  uint8_t hdr[WAV_HDR_SIZE];
  createWavHeader(hdr, 0);
  f.write(hdr, WAV_HDR_SIZE);
  int16_t buf[BUF_SIZE];
  size_t bytesRead = 0, total = 0;
  uint32_t start = millis(), recDur = RECORD_TIME * 1000, last = start;
  Serial.println("Recording...");
  while(millis() - start < recDur) {
    if(millis() - last >= 1000) {
      Serial.printf("Remaining: %d sec\n", (recDur - (millis() - start)) / 1000);
      last = millis();
    }
    if(i2s_read(I2S_PORT, buf, sizeof(buf), &bytesRead, portMAX_DELAY) != ESP_OK) { Serial.println("I2S read error"); continue; }
    if(bytesRead > 0) {
      size_t wr = f.write((uint8_t*)buf, bytesRead);
      if(wr != bytesRead) Serial.printf("Write warn: %d/%d\n", wr, bytesRead);
      total += wr;
    }
  }
  f.seek(0);
  createWavHeader(hdr, total);
  f.write(hdr, WAV_HDR_SIZE);
  f.close();
  Serial.printf("Recorded %u bytes to %s\n", total, FILENAME);
  return total > 0;
}

bool checkFile() {
  if(!SPIFFS.exists(FILENAME)) { Serial.printf("File %s not found\n", FILENAME); return false; }
  File f = SPIFFS.open(FILENAME, "r");
  if(!f) { Serial.printf("Open file %s error\n", FILENAME); return false; }
  size_t sz = f.size();
  Serial.printf("File size: %u bytes\n", sz);
  if(sz <= WAV_HDR_SIZE) { Serial.println("File too small"); f.close(); return false; }
  uint8_t hdr[WAV_HDR_SIZE];
  if(f.readBytes((char*)hdr, WAV_HDR_SIZE) != WAV_HDR_SIZE) { Serial.println("Header read error"); f.close(); return false; }
  if(!(hdr[0]=='R' && hdr[1]=='I' && hdr[2]=='F' && hdr[3]=='F')) { Serial.println("Bad RIFF"); f.close(); return false; }
  if(!(hdr[8]=='W' && hdr[9]=='A' && hdr[10]=='V' && hdr[11]=='E')) { Serial.println("Bad WAVE"); f.close(); return false; }
  uint32_t dSize = hdr[40] | (hdr[41] << 8) | (hdr[42] << 16) | (hdr[43] << 24);
  Serial.printf("Data size: %u bytes\n", dSize);
  int16_t samples[10];
  f.seek(WAV_HDR_SIZE);
  size_t r = f.readBytes((char*)samples, sizeof(samples));
  bool nonZero = false;
  for (size_t i = 0; i < r / sizeof(int16_t); i++) { if(samples[i] != 0) { nonZero = true; break; } }
  if(!nonZero) Serial.println("Warning: Silence");
  f.close();
  return true;
}

void setup(){
  Serial.begin(115200);
  delay(1000);
  Serial.println("\nESP32 I2S Recorder");
  if(!SPIFFS.begin(true)) { Serial.println("SPIFFS init fail"); while(1) delay(1000); }
  Serial.printf("SPIFFS: Total: %u, Used: %u\n", SPIFFS.totalBytes(), SPIFFS.usedBytes());
  listFiles();
  if(SPIFFS.exists(FILENAME)) Serial.printf("File %s will be overwritten.\n", FILENAME);
  Serial.println("I2S init...");
  i2s_init();
  delay(500);
  Serial.println("Recording...");
  bool s = recordAudio();
  if(s) {
    Serial.println("Recording successful.");
    listFiles();
    Serial.println("Checking file...");
    if(checkFile()) Serial.println("File valid.");
    else Serial.println("File error.");
  } else Serial.println("Recording failed.");
  i2s_driver_uninstall(I2S_PORT);
  Serial.println("I2S driver uninstalled.");
  Serial.println("Processing complete. Reset board to record again.");
}

void loop() {
  delay(1000);
}
