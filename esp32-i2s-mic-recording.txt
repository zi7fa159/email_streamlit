#include <Arduino.h>
#include <driver/i2s.h>
#include <SPIFFS.h>

// I2S configuration
#define I2S_PORT I2S_NUM_0
#define I2S_WS 15      // LRC/WS
#define I2S_SCK 14     // SCK
#define I2S_SD 32      // SD
#define SAMPLE_RATE 16000
#define SAMPLE_BITS 16
#define WAV_HEADER_SIZE 44

// Recording parameters
#define RECORD_TIME 5  // seconds
#define BUFFER_SIZE 512

// Basic WAV header creation
void createWavHeader(byte* header, uint32_t wavSize) {
  // RIFF chunk
  header[0] = 'R'; header[1] = 'I'; header[2] = 'F'; header[3] = 'F';
  uint32_t fileSize = wavSize + 36;
  header[4] = fileSize & 0xFF; header[5] = (fileSize >> 8) & 0xFF;
  header[6] = (fileSize >> 16) & 0xFF; header[7] = (fileSize >> 24) & 0xFF;
  
  // WAVE chunk
  header[8] = 'W'; header[9] = 'A'; header[10] = 'V'; header[11] = 'E';
  
  // fmt subchunk
  header[12] = 'f'; header[13] = 'm'; header[14] = 't'; header[15] = ' ';
  header[16] = 16; header[17] = 0; header[18] = 0; header[19] = 0; // Subchunk size (16 bytes)
  header[20] = 1; header[21] = 0; // PCM format = 1
  header[22] = 1; header[23] = 0; // Mono = 1 channel
  
  // Sample rate
  header[24] = SAMPLE_RATE & 0xFF;
  header[25] = (SAMPLE_RATE >> 8) & 0xFF;
  header[26] = (SAMPLE_RATE >> 16) & 0xFF;
  header[27] = (SAMPLE_RATE >> 24) & 0xFF;
  
  // Byte rate
  uint32_t byteRate = SAMPLE_RATE * 2;
  header[28] = byteRate & 0xFF;
  header[29] = (byteRate >> 8) & 0xFF;
  header[30] = (byteRate >> 16) & 0xFF;
  header[31] = (byteRate >> 24) & 0xFF;
  
  header[32] = 2; header[33] = 0; // Block align
  header[34] = SAMPLE_BITS; header[35] = 0; // Bits per sample
  
  // data subchunk
  header[36] = 'd'; header[37] = 'a'; header[38] = 't'; header[39] = 'a';
  header[40] = wavSize & 0xFF;
  header[41] = (wavSize >> 8) & 0xFF;
  header[42] = (wavSize >> 16) & 0xFF;
  header[43] = (wavSize >> 24) & 0xFF;
}

// Initialize I2S driver
void i2s_init() {
  i2s_config_t i2s_config = {
    .mode = (i2s_mode_t)(I2S_MODE_MASTER | I2S_MODE_RX),
    .sample_rate = SAMPLE_RATE,
    .bits_per_sample = (i2s_bits_per_sample_t)SAMPLE_BITS,
    .channel_format = I2S_CHANNEL_FMT_ONLY_LEFT,
    .communication_format = I2S_COMM_FORMAT_I2S,
    .intr_alloc_flags = ESP_INTR_FLAG_LEVEL1,
    .dma_buf_count = 4,     // Using 4 as in the working code
    .dma_buf_len = BUFFER_SIZE,
    .use_apll = false
  };
  
  i2s_pin_config_t pin_config = {
    .bck_io_num = I2S_SCK,
    .ws_io_num = I2S_WS,
    .data_out_num = I2S_PIN_NO_CHANGE,
    .data_in_num = I2S_SD
  };
  
  // First uninstall driver if it was already installed
  i2s_driver_uninstall(I2S_PORT);
  
  // Install and configure I2S driver
  esp_err_t err = i2s_driver_install(I2S_PORT, &i2s_config, 0, NULL);
  if (err != ESP_OK) {
    Serial.printf("Failed to install I2S driver: %d\n", err);
    return;
  }
  
  err = i2s_set_pin(I2S_PORT, &pin_config);
  if (err != ESP_OK) {
    Serial.printf("Failed to set I2S pins: %d\n", err);
    return;
  }
  
  i2s_start(I2S_PORT);
}

// List all files in SPIFFS
void listSPIFFSFiles() {
  File root = SPIFFS.open("/");
  File file = root.openNextFile();
  Serial.println("SPIFFS files:");
  while (file) {
    Serial.printf("  %s (%u bytes)\n", file.name(), file.size());
    file = root.openNextFile();
  }
}

// Record audio to SPIFFS
bool recordAudio(const char* filename) {
  // Open file
  File file = SPIFFS.open(filename, "w");
  if (!file) {
    Serial.printf("Failed to open file: %s\n", filename);
    Serial.println("Trying with default filename...");
    file = SPIFFS.open("/recording.wav", "w");
    if (!file) {
      Serial.println("Failed to open default file");
      return false;
    }
  }
  
  // Create WAV header (will update later)
  byte header[WAV_HEADER_SIZE];
  createWavHeader(header, 0);
  file.write(header, WAV_HEADER_SIZE);
  
  // Record audio
  int16_t buffer[BUFFER_SIZE];
  size_t bytesIn = 0;
  size_t totalBytes = 0;
  uint32_t recordTime = RECORD_TIME * 1000;
  uint32_t startTime = millis();
  uint32_t lastReport = startTime;
  
  Serial.println("Recording...");
  while (millis() - startTime < recordTime) {
    // Print progress every second
    if (millis() - lastReport >= 1000) {
      Serial.printf("Recording: %d seconds remaining...\n", 
                   (recordTime - (millis() - startTime)) / 1000);
      lastReport = millis();
    }
    
    // Using portMAX_DELAY from the working code
    esp_err_t result = i2s_read(I2S_PORT, buffer, sizeof(buffer), &bytesIn, portMAX_DELAY);
    
    if (result != ESP_OK) {
      Serial.printf("Error reading from I2S: %d\n", result);
      continue;
    }
    
    if (bytesIn > 0) {
      size_t bytesWritten = file.write((uint8_t*)buffer, bytesIn);
      if (bytesWritten != bytesIn) {
        Serial.printf("Warning: Wrote only %d of %d bytes\n", bytesWritten, bytesIn);
      }
      totalBytes += bytesWritten;
    }
  }
  
  // Update WAV header with correct size
  file.seek(0);
  createWavHeader(header, totalBytes);
  file.write(header, WAV_HEADER_SIZE);
  file.close();
  
  Serial.printf("Recorded %u bytes to %s\n", totalBytes, filename);
  return (totalBytes > WAV_HEADER_SIZE);  // Return success only if we recorded actual data
}

// Check if the recorded file is valid
bool checkRecordedFile(const char* filename) {
  if (!SPIFFS.exists(filename)) {
    Serial.printf("File %s not found\n", filename);
    return false;
  }
  
  File file = SPIFFS.open(filename, "r");
  if (!file) {
    Serial.printf("Failed to open file %s for checking\n", filename);
    return false;
  }
  
  // Check file size
  size_t fileSize = file.size();
  Serial.printf("File size: %u bytes\n", fileSize);
  
  if (fileSize <= WAV_HEADER_SIZE) {
    Serial.println("File too small, contains only header");
    file.close();
    return false;
  }
  
  // Check WAV header
  byte header[WAV_HEADER_SIZE];
  size_t bytesRead = file.readBytes((char*)header, WAV_HEADER_SIZE);
  
  if (bytesRead != WAV_HEADER_SIZE) {
    Serial.println("Failed to read WAV header");
    file.close();
    return false;
  }
  
  // Check RIFF signature
  if (header[0] != 'R' || header[1] != 'I' || header[2] != 'F' || header[3] != 'F') {
    Serial.println("Invalid RIFF signature");
    file.close();
    return false;
  }
  
  // Check WAVE signature
  if (header[8] != 'W' || header[9] != 'A' || header[10] != 'V' || header[11] != 'E') {
    Serial.println("Invalid WAVE signature");
    file.close();
    return false;
  }
  
  // Get data size from header
  uint32_t dataSize = header[40] | (header[41] << 8) | (header[42] << 16) | (header[43] << 24);
  Serial.printf("Data size from header: %u bytes\n", dataSize);
  
  // Read first few samples to check for non-zero values
  int16_t samples[10];
  file.seek(WAV_HEADER_SIZE);
  bytesRead = file.readBytes((char*)samples, sizeof(samples));
  
  bool hasNonZero = false;
  for (int i = 0; i < bytesRead / 2; i++) {
    if (samples[i] != 0) {
      hasNonZero = true;
      break;
    }
  }
  
  if (!hasNonZero) {
    Serial.println("Warning: File may contain only silence (zeros)");
  }
  
  file.close();
  return true;
}

void setup() {
  Serial.begin(115200);
  delay(1000); // Give serial monitor time to start
  
  Serial.println("\nESP32 I2S Microphone Recorder");
  
  // Always format SPIFFS at startup as requested
  Serial.println("Formatting SPIFFS...");
  SPIFFS.format();
  
  // Initialize SPIFFS
  if (!SPIFFS.begin(true)) {
    Serial.println("SPIFFS init failed after formatting. Halting.");
    while (1) delay(1000);
  }
  
  Serial.println("SPIFFS formatted and mounted successfully.");
  
  // Show SPIFFS info
  Serial.printf("SPIFFS: Total: %u bytes, Used: %u bytes\n", 
                SPIFFS.totalBytes(), SPIFFS.usedBytes());
  
  // List existing files (should be empty after format)
  listSPIFFSFiles();
  
  // Use simple filename
  const char* filename = "/rec.wav";
  
  // Initialize I2S with debug output
  Serial.println("Initializing I2S...");
  i2s_init();
  Serial.println("I2S initialized.");
  
  // Add a delay before recording to stabilize
  delay(500);
  
  // Record audio
  Serial.println("Starting recording...");
  bool success = recordAudio(filename);
  
  // Show results
  if (success) {
    Serial.println("Recording successful.");
    listSPIFFSFiles();
    
    // Check if file is valid
    Serial.println("Checking recorded file...");
    if (checkRecordedFile(filename)) {
      Serial.println("File check successful. WAV file is valid.");
      
      // Optional: Read and print some samples
      File file = SPIFFS.open(filename, "r");
      if (file) {
        file.seek(WAV_HEADER_SIZE);
        int16_t samples[20];
        size_t bytesRead = file.readBytes((char*)samples, sizeof(samples));
        Serial.println("First few samples:");
        for (int i = 0; i < bytesRead / 2; i++) {
          Serial.printf("%d ", samples[i]);
        }
        Serial.println();
        file.close();
      }
    } else {
      Serial.println("File check failed. WAV file may be corrupted.");
    }
  } else {
    Serial.println("Recording failed.");
    Serial.println("Troubleshooting tips:");
    Serial.println("1. Check microphone connections");
    Serial.println("2. Verify I2S pin definitions match your hardware");
    Serial.println("3. Try different I2S settings");
    Serial.println("4. Check power supply stability");
  }
  
  // Cleanup
  i2s_driver_uninstall(I2S_PORT);
  Serial.println("I2S driver uninstalled.");
  
  Serial.println("Processing complete. Reset the board to record again.");
}

void loop() {
  // Empty - nothing happens here, waiting for user to reset the board
  delay(1000);
}
